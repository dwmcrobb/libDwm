\newcommand{\SizedLengthExample}[3]{
\centering
\noindent\begin{tikzpicture}[
    start chain = A going right,
    node distance = 0pt, % Ensure no gap between bit nodes
    bit node/.style = {draw, minimum height=20pt, minimum width=\textwidth/72, text width=\textwidth/72, align=center, font=\ttfamily, inner sep=0pt, outer sep=0pt, on chain=A},
    byte node/.style = {draw, minimum height=20pt, minimum width=\dimexpr\textwidth/9\relax, text width=\dimexpr\textwidth/8\relax, align=center, font=\ttfamily, inner sep=0pt, outer sep=0pt, on chain=A}
  ]
  \def\mybitstring{\texttt{1},{},{},{},{},0,1,0}
  \def\mybytestring{0xAA,0xBB,0xCC}
  % Draw the bit nodes and indices
  \foreach \bit [count=\i] in {#1} {
    % \ifnum\bit=1 \def\bitcolor{black!20} \else \def\bitcolor{white} \fi
    \numtestTF{ 1 < \i < 6 }
              { \node[bit node, fill=black!20] (bit-\i) {};
                \node[below=0mm of bit-\i, font=\tiny] {\i}; }
              { \node[bit node, fill=white,font=\small] (bit-\i) {\bit};
                \node[below=0mm of bit-\i, font=\tiny] {\i}; }
  }
  
  % Add overbrace for a specific section
  \draw [decorate, decoration={brace, amplitude=5pt}] (bit-1.north west) -- (bit-1.north east) 
  node [above=5pt, midway, font=\footnotesize] {Endian};
  
  % Add overbrace for the rest of the bits
  \draw [decorate, decoration={brace, amplitude=5pt}] (bit-6.north west) -- (bit-8.north east) 
  node [above=5pt, midway, font=\footnotesize] {Size};

  % Add label for the ES byte
  \draw [decorate, decoration={brace, mirror, amplitude=0pt}] (bit-1.south west) -- (bit-8.south east)  
  node [below=7pt, midway, font=\footnotesize] {ES byte};

  % Draw the byte nodes and indices
  \foreach \byte [count=\i] in {#2} {
    % \ifnum\bit=1 \def\bitcolor{black!20} \else \def\bitcolor{white} \fi
    \node[byte node, fill=white] (byte-\i) {\byte};
    \node[below=0mm of byte-\i, font=\tiny] {\i};
  }
\end{tikzpicture}
\captionof{figure}{#3}
}
