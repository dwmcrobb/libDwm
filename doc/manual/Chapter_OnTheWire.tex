\subsection{\texttt{Dwm::SizedLength} encoding}\label{encoding:DwmSizedLength}

\LengthEncodedData{uint8\_t}{\texttt{value}}{\texttt{Dwm::SizedLength} encoding}{Encoding byte}

\subsubsection{Encoding byte}

\SizedLengthExample{{},{},{},{},{},{},{},{}}{}{Encoding byte}

The encoding byte uses 4 of the available 8 bits.  The \texttt{Endian} bit
indicates endiannness (0 = little endian, 1 = big endian) of the value that
follows the encoding byte.  The \texttt{Size} bits indicate the length of
the value (in bytes) following the encoding byte, minus 1:

  $length\ of\ \texttt{value} = \texttt{Size} + 1$

The remaining bits in the encoding byte are unused.

The \texttt{value} of a \texttt{Dwm::SizedLength} is always written in
native byte order (typically little-endian).  The read member functions
of \texttt{Dwm::SizedLength} will convert endianness if needed.

\subsubsection{Examples}
\SizedLengthExample{1,{},{},{},{},0,0,0}{0xAA}{One byte value \texttt{0xAA}, big-endian.}

\SizedLengthExample{0,{},{},{},{},0,0,0}{0xAA}{One byte value \texttt{0xAA}, little-endian.}

\SizedLengthExample{1,{},{},{},{},0,0,1}{0xAA,0xBB}{Two byte value \texttt{0xAABB}, big-endian.}

\SizedLengthExample{0,{},{},{},{},0,0,1}{0xBB,0xAA}{Two byte value \texttt{0xAABB}, little-endian.}

\SizedLengthExample{1,{},{},{},{},0,1,0}{0xAA,0xBB,0xCC}{Three byte value \texttt{0xAABBCC}, big-endian.}

\SizedLengthExample{0,{},{},{},{},0,1,0}{0xCC,0xBB,0xAA}{Three byte value \texttt{0xAABBCC}, little-endian.}

\SizedLengthExample{1,{},{},{},{},0,1,1}{0xAA,0xBB,0xCC,0xDD}{Four byte value \texttt{0xAABBCCDD}, big-endian.}

\SizedLengthExample{0,{},{},{},{},0,1,1}{0xDD,0xCC,0xBB,0xAA}{Four byte value \texttt{0xAABBCCDD}, little-endian.}

\SizedLengthExample{1,{},{},{},{},1,0,0}{0xAA,0xBB,0xCC,0xDD,0xEE}{Five byte value \texttt{0xAABBCCDDEE}, big-endian.}

\SizedLengthExample{0,{},{},{},{},1,0,0}{0xEE,0xDD,0xCC,0xBB,0xAA}{Five byte value \texttt{0xAABBCCDDEE}, little-endian.}

\SizedLengthExample{1,{},{},{},{},1,0,1}{0xAA,0xBB,0xCC,0xDD,0xEE,0xFF}{Six byte value \texttt{0xAABBCCDDEEFF}, big-endian.}

\SizedLengthExample{0,{},{},{},{},1,0,1}{0xFF,0xEE,0xDD,0xCC,0xBB,0xAA}{Six byte value \texttt{0xAABBCCDDEEFF}, little-endian.}

\SizedLengthExample{1,{},{},{},{},1,1,0}{0xAA,0xBB,0xCC,0xDD,0xEE,0xFF,0x99}{Seven byte value \texttt{0xAABBCCDDEEFF99}, big-endian.}

\SizedLengthExample{0,{},{},{},{},1,1,0}{0x99,0xFF,0xEE,0xDD,0xCC,0xBB,0xAA}{Seven byte value \texttt{0xAABBCCDDEEFF99}, little-endian.}

\SizedLengthExample{1,{},{},{},{},1,1,1}{0xAA,0xBB,0xCC,0xDD,0xEE,0xFF,0x99,0x88}{Eight byte value \texttt{0xAABBCCDDEEFF9988}, big-endian.}

\SizedLengthExample{0,{},{},{},{},1,1,1}{0x88,0x99,0xFF,0xEE,0xDD,0xCC,0xBB,0xAA}{Eight byte value \texttt{0xAABBCCDDEEFF9988}, little-endian.}
    
\subsection{\texttt{std::string} encoding}
\texttt{std::string} is encoded with a 64-bit length field and then
the characters in the string.

\LengthEncodedData{uint64\_t}{\texttt{s[0] ... s[length-1]}}{\texttt{std::string} encoding}{length}

\dwmNote{The \texttt{length} field is likely to be changed to use
  \texttt{Dwm::SizedLength} in the future.}

\subsection{\texttt{std::optional<T>} encoding}\label{encoding:stdOptional}
A \texttt{std::optional<T>} is encoded with a single byte to indicate whether
or not the value is present, then the value if present.

\LengthEncodedData{uint8\_t}{\texttt{T}}{\texttt{std::optional<T>} encoding}{value present?}

\subsection{\texttt{std::pair<class FirstType, class SecondType>} encoding}\label{encoding:stdPair}
The encoding of a \texttt{std::pair} is as you'd expect: the first value
followed by the second value.

\StdPairData{FirstType}{SecondType}{\texttt{std::pair<class FirstType, class SecondType>} encoding}

\subsection{\texttt{std::vector<class T>} encoding}\label{encoding:stdVector}
A \texttt{std::vector} is encoded with a 64-bit \texttt{length} field
followed by \texttt{length} elements of the \texttt{vector}.

\LengthEncodedData{uint64\_t}{\texttt{$T_0$  ...  $T_{length-1}$}}{\texttt{std::vector<T>} encoding}{length}

\dwmNote{The \texttt{length} field is likely to be changed to use
\texttt{Dwm::SizedLength} encoding in the future.}
